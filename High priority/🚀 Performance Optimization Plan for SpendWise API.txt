### **Issue Analysis**
Login is slow because it:
1. Hashes the password (bcrypt)
2. Queries the database for the user
3. Creates JWT tokens
4. Sets cookies

#### **1.2 Implement Redis Session Caching** (2 hours)
```python
# app/services/cache_service.py - Add this method
@classmethod
async def cache_user_session(cls, user_id: str, session_data: dict, ttl: int = 3600):
    """Cache user session data"""
    if cls._redis_client:
        key = f"user_session:{user_id}"
        await cls._redis_client.setex(key, ttl, json.dumps(session_data))

@classmethod
async def get_cached_user(cls, user_id: str):
    """Get cached user data"""
    if cls._redis_client:
        key = f"user_session:{user_id}"
        data = await cls._redis_client.get(key)
        return json.loads(data) if data else None
```

#### **1.4 Add Login Attempt Tracking with Redis**
```python
# app/api/v1/endpoints/auth.py - Optimized login
@router.post("/login")
async def login(login_data: UserLogin, response: Response, db: AsyncSession = Depends(get_db)):
    # Check Redis for failed attempts
    attempts_key = f"login_attempts:{login_data.email}"
    attempts = await redis.get(attempts_key)
    
    if attempts and int(attempts) > 5:
        raise HTTPException(429, "Too many attempts. Try again later.")
    
    # Query with index (fast)
    result = await db.execute(
        select(User).where(User.email == login_data.email)
    )
    user = result.scalars().first()
    
    # Verify password (this is the slow part)
    if not user or not verify_password(login_data.password, user.hashed_password):
        await redis.incr(attempts_key)
        await redis.expire(attempts_key, 900)  # 15 minutes
        raise HTTPException(401, "Invalid credentials")
    
    # Clear attempts on success
    await redis.delete(attempts_key)
    
    # Create session
    # ... rest of login logic
```

### **2.2 Implement Redis Caching for Read-Heavy Endpoints**

Create a caching service:

```python
# app/services/cache_service.py - Enhanced version
import json
from typing import Optional, Any
import redis.asyncio as redis
from app.core.config import settings

class CacheService:
    _redis_client = None
    
    @classmethod
    async def get_redis(cls):
        if not cls._redis_client:
            cls._redis_client = await redis.from_url(
                f"redis://{settings.REDIS_HOST}:{settings.REDIS_PORT}/{settings.REDIS_DB}",
                decode_responses=True
            )
        return cls._redis_client
    
    @classmethod
    async def get_or_compute(cls, key: str, ttl: int, compute_func):
        """Get from cache or compute and cache"""
        redis = await cls.get_redis()
        
        # Try cache first
        cached = await redis.get(key)
        if cached:
            return json.loads(cached)
        
        # Compute and cache
        result = await compute_func()
        await redis.setex(key, ttl, json.dumps(result, default=str))
        return result
    
    @classmethod
    async def invalidate_pattern(cls, pattern: str):
        """Invalidate all keys matching pattern"""
        redis = await cls.get_redis()
        keys = await redis.keys(pattern)
        if keys:
            await redis.delete(*keys)
```

### **2.3 Add Caching to Endpoints**

```python
# app/api/v1/endpoints/categories.py - Example with caching
from app.services.cache_service import CacheService

@router.get("/", response_model=List[CategoryOut])
async def read_categories(
    skip: int = 0,
    limit: int = 100,
    category_type: Optional[CategoryType] = Query(None),
    current_user = Depends(get_current_user),
    session: AsyncSession = Depends(get_db)
) -> List[CategoryOut]:
    # Create cache key based on user and filters
    cache_key = f"categories:{current_user.id}:{skip}:{limit}:{category_type}"
    
    async def fetch_categories():
        return await CategoryService.get_categories(
            session=session,
            user_id=current_user.id,
            skip=skip,
            limit=limit,
            category_type=category_type
        )
    
    # Cache for 5 minutes (300 seconds)
    return await CacheService.get_or_compute(cache_key, 300, fetch_categories)
```

## 🚀 **PHASE 5: Frontend Optimizations**

### **5.1 Add Request Debouncing for Search**

```typescript
// frontend/src/hooks/useDebounce.ts
import { useState, useEffect } from 'react';

export function useDebounce<T>(value: T, delay: number): T {
  const [debouncedValue, setDebouncedValue] = useState<T>(value);

  useEffect(() => {
    const handler = setTimeout(() => {
      setDebouncedValue(value);
    }, delay);

    return () => {
      clearTimeout(handler);
    };
  }, [value, delay]);

  return debouncedValue;
}
```
